cmake_minimum_required(VERSION 3.22.1)
project(kanade_nac LANGUAGES CXX)

file(TO_CMAKE_PATH "${KANADE_NAC_MODEL}" KANADE_NAC_MODEL)
file(SHA256 "${KANADE_NAC_MODEL}" model_hash)
if(NOT model_hash STREQUAL "592851ab140c0b4a99bceb0550c214d4ca8c05ba4378a4b16419be1728f00cfa")
    message(FATAL_ERROR "KANADE_NAC_MODEL must be the 7187332.mnn NAC decoder model")
endif()
file(SIZE "${KANADE_NAC_MODEL}" model_size)
if(WIN32)
    enable_language(RC)
    set(model_source nac_model.rc)
else()
    enable_language(ASM)
    set(model_source nac_model.S)
endif()
configure_file("${model_source}.in" "${model_source}" @ONLY)
set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/${model_source}" PROPERTY OBJECT_DEPENDS "${KANADE_NAC_MODEL}")

set(mnn_dir "${CMAKE_CURRENT_BINARY_DIR}/mnn-3.6.2")
file(MAKE_DIRECTORY "${mnn_dir}/include/MNN")
function(fetch_mnn url output hash)
    file(DOWNLOAD "${url}" "${output}" EXPECTED_HASH "SHA256=${hash}" TLS_VERIFY ON TIMEOUT 180)
endfunction()

function(mnn_header name hash)
    fetch_mnn("https://raw.githubusercontent.com/alibaba/MNN/3.6.2/include/MNN/${name}"
        "${mnn_dir}/include/MNN/${name}" "${hash}")
endfunction()
mnn_header(Interpreter.hpp 3a8f13170819eb702eec30200d53febe908337577756601336e3782cdac0d7ad)
mnn_header(Tensor.hpp 6f28b4d0338c6760582ed8d5632fa0c3691bdfacc421753b6d86407466a7bf5c)
mnn_header(MNNForwardType.h 7a3aa7951b4999d579c0185d9f2b50698595c80d5116ed23e735821493d57153)
mnn_header(MNNDefine.h c20693fb9a7d0e13c1868a9fc436154a5d0bc73e168255a3b95a3eb25ac1b97a)
mnn_header(ErrorCode.hpp de7d2290e4e2f024ae51707298c30721a283367eb90a7abac113d27248d56dd0)
mnn_header(HalideRuntime.h f1d35170fa3fc1c7433ca1d18ca5d016ad230f26a0ecfba91b9a7ef9a49f7a37)

add_library(MNN SHARED IMPORTED)
if(WIN32)
    include(mnn_windows.cmake)
    set_target_properties(MNN PROPERTIES
        IMPORTED_LOCATION "${mnn_dir}/MNN.dll"
        IMPORTED_IMPLIB "${mnn_dir}/MNN.lib")
else()
    set(mnn_release "mnn_3.6.2_android_armv7_armv8_cpu_opencl_vulkan")
    set(mnn_lib "${mnn_dir}/${mnn_release}/arm64-v8a/libMNN.so")
    if(NOT EXISTS "${mnn_lib}")
        fetch_mnn("https://github.com/alibaba/MNN/releases/download/3.6.2/${mnn_release}.zip"
            "${mnn_dir}/mnn.zip" 0b4cc255964de1e3ac8dce75dce316699a0a28b914ecacb7bea8465666858ece)
        file(ARCHIVE_EXTRACT INPUT "${mnn_dir}/mnn.zip" DESTINATION "${mnn_dir}"
            PATTERNS "${mnn_release}/arm64-v8a/libMNN.so")
    endif()
    set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${mnn_lib}")
endif()

add_library(kanade_nac SHARED nac_decoder.cpp "${CMAKE_CURRENT_BINARY_DIR}/${model_source}")
target_include_directories(kanade_nac PRIVATE "${mnn_dir}/include")
target_compile_features(kanade_nac PRIVATE cxx_std_17)
target_compile_definitions(kanade_nac PRIVATE "KANADE_NAC_MODEL_SIZE=${model_size}")
target_link_libraries(kanade_nac PRIVATE MNN)
if(WIN32)
    set_property(TARGET kanade_nac PROPERTY KANADE_MNN_RUNTIME "${mnn_dir}/MNN.dll")
endif()
if(MSVC)
    # The official MNN DLL uses the release CRT and STL ABI in every build mode.
    set_property(TARGET kanade_nac PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    target_compile_definitions(kanade_nac PRIVATE _ITERATOR_DEBUG_LEVEL=0 NOMINMAX)
    target_compile_options(kanade_nac PRIVATE /utf-8)
endif()
