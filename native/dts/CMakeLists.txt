file(TO_CMAKE_PATH "${KANADE_DTS_ANDROID_RUNTIME}" runtime)
set(sdk "${runtime}/libDTSPlaybackSDK.so")
set(config "${runtime}/dts_config.yaml")
set(license "${runtime}/dts_hpx.lic")
foreach(input IN ITEMS "${sdk}" "${config}" "${license}")
    if(NOT EXISTS "${input}")
        message(FATAL_ERROR "Missing local DTS Playback SDK input ${input}")
    endif()
endforeach()
file(SHA256 "${sdk}" sdk_hash)
if(NOT sdk_hash STREQUAL "71b9872cca9e031b293eef7ec8a5762cf1069ad8762a9e17f261d20d1bff95d1")
    message(FATAL_ERROR "Unsupported ARM64 DTS Playback SDK binary")
endif()
file(SIZE "${config}" config_size)
file(SIZE "${license}" license_size)

add_library(DTSPlaybackSDK SHARED IMPORTED)
set_target_properties(DTSPlaybackSDK PROPERTIES IMPORTED_LOCATION "${sdk}")

configure_file(dts_assets.S.in dts_assets.S @ONLY)
set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/dts_assets.S" PROPERTY OBJECT_DEPENDS "${config};${license}")
add_library(kanade_dts SHARED dts_decoder.cpp "${CMAKE_CURRENT_BINARY_DIR}/dts_assets.S")
target_compile_features(kanade_dts PRIVATE cxx_std_17)
target_compile_definitions(kanade_dts PRIVATE "DTS_CONFIG_SIZE=${config_size}" "DTS_LICENSE_SIZE=${license_size}")
target_link_libraries(kanade_dts PRIVATE DTSPlaybackSDK)
